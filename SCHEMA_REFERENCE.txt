                   9. QUICK REFERENCE QUERIES
================================================================================

GET LATEST VALUATION FOR A STARTUP:
  SELECT * FROM valuations
  WHERE startup_id = '{startup_id}'
  AND is_latest = true;

GET ALL METHODS FOR A VALUATION:
  SELECT method_name, low_estimate, mid_estimate, high_estimate
  FROM valuation_methods
  WHERE valuation_id = '{valuation_id}'
  ORDER BY method_name;

GET USER'S STARTUP COUNT:
  SELECT COUNT(*) as startup_count
  FROM startups
  WHERE user_id = '{user_id}'
  AND status = 'active';

GET VALUATION HISTORY FOR STARTUP:
  SELECT v.blended_weighted_average, v.created_at
  FROM valuations v
  WHERE v.startup_id = '{startup_id}'
  ORDER BY v.created_at DESC
  LIMIT 12;

GET PENDING EMAIL SEQUENCE EMAILS:
  SELECT * FROM email_sequence_leads
  WHERE day_3_sent_at IS NULL
  AND day_3_scheduled_for <= NOW();

GET USER'S SUBSCRIPTION STATUS:
  SELECT plan, plan_active, subscription_start_date, subscription_end_date
  FROM users
  WHERE id = '{user_id}';

GET VALUATION CONFIDENCE & COMPLETENESS:
  SELECT confidence_level, data_completeness, blended_weighted_average
  FROM valuations
  WHERE startup_id = '{startup_id}'
  AND is_latest = true;

COUNT REPORTS GENERATED:
  SELECT report_type, COUNT(*) as count
  FROM reports
  WHERE created_at >= NOW() - INTERVAL '30 days'
  GROUP BY report_type;
